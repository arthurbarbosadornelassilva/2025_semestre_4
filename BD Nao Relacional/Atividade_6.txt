2)
a)
MATCH (a:Artista)
WHERE a.altura IS NOT NULL
  AND a.local_nascto IS NOT NULL
RETURN a.local_nascto AS Local_Nascimento,
  MAX(a.altura) AS Maior_Altura
  ORDER BY Maior_Altura

b)
MATCH (a:Artista)
WITH a,
CASE 
WHEN a.dt_nascto =~ '\\d{4}-\\d{2}-\\d{2}' THEN date(a.dt_nascto)
WHEN a.dt_nascto =~ '\\d{4}' THEN date({year: toInteger(a.dt_nascto)})
ELSE NULL
END AS dataFinal
WITH dataFinal.year as ano_nascto
WHERE ano_nasct IS NOT NULL
RETURN ano_nascto, COUNT(*) AS contagem_ano_nascto
ORDER BY contagem_ano_nascto DESC

c)
MATCH(a:Artista)
WITH a,
CASE 
WHEN a.dt_nascto =~ '\\d{4}-\\d{2}-\\d{2}' THEN date(a.dt_nascto)
WHEN a.dt_nascto =~ '\\d{4}' THEN date({year: toInteger(a.dt_nascto)})
ELSE NULL
END AS dataFinal
WITH dataFinal.year AS ano_nascto, min(a.altura) AS menor_altura, a.nome as nome
WHERE ano_nascto IS NOT NULL AND menor_altura IS NOT NULL
RETURN ano_nascto, menor_altura, nome
ORDER BY ano_nasct
